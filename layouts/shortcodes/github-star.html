{{- /* Reusable GitHub star callout. {{< github-star repo="furkankoykiran/Leadwalk" >}} The star count is read from the GitHub REST API at build time. The request is cached by Hugo for the rest of the build, never happens in the browser, and a rate limit or network error only hides the count; the callout itself always renders. Set HUGO_GITHUB_TOKEN in CI for a higher rate limit. */ -}} {{- $repo := .Get "repo" | default (.Get 0) -}} {{- if not $repo -}} {{- errorf "github-star: the repo parameter is required (%s)" .Position -}} {{- end -}} {{- $repo = strings.TrimPrefix "https://github.com/" $repo -}} {{- $repo = strings.TrimPrefix "http://github.com/" $repo -}} {{- $repo = strings.TrimSuffix "/" $repo -}} {{- $name := path.Base $repo -}} {{- $stars := 0 -}} {{- $known := false -}} {{- $headers := dict "Accept" "application/vnd.github+json" "User-Agent" "furkankoykiran-blog" -}} {{- with os.Getenv "HUGO_GITHUB_TOKEN" -}} {{- $headers = merge $headers (dict "Authorization" (printf "Bearer %s" .)) -}} {{- end -}} {{- $api := printf "https://api.github.com/repos/%s" $repo -}} {{- with try (resources.GetRemote $api (dict "headers" $headers "timeout" "5s")) -}} {{- with .Err -}} {{- /* A missing token or a rate limit must not fail the build (lint runs with --panicOnWarning). Only surface the reason in local development. */ -}} {{- if eq (os.Getenv "HUGO_ENVIRONMENT") "development" -}} {{- warnf "github-star: cannot read %s: %s" $repo . -}} {{- end -}} {{- else -}} {{- with .Value -}} {{- $data := .Content | transform.Unmarshal -}} {{- with $data.stargazers_count -}} {{- $stars = int . -}} {{- $known = true -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}}
